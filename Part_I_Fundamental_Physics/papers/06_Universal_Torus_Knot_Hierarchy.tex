\begin{abstract}
Papers~I--V of the Constraint-Driven Field Theory (CDFT) series built the
lepton-sector construction: the electromagnetic fine-structure constant
$\alpha \equiv e^2/(\hbar c) = 1/137.036$ from vortex ring equilibrium, the Koide
formula from $\mathbb{Z}_3$ symmetry, and all three lepton masses from a single
input ($m_e$).
This paper extends the framework beyond leptons by studying the full hierarchy
of torus knot families $T(2,n)$ for odd $n \geq 3$.

We prove two new theorems.
\textbf{Theorem~1} (universal coefficient): the Brannen amplitude coefficient
$c_n = \sqrt{2}$ for all $\mathbb{Z}_n$ families, fixed by a maximum-entropy
self-consistency condition on the amplitude distribution.
\textbf{Theorem~2} ($\mathbb{Z}_3$ uniqueness): $\mathbb{Z}_3$ is the unique
torus knot family for which the constant Koide-like invariant
$Q_n = 2/n$ and the $\mathbb{Z}_n$ self-consistency condition hold
simultaneously on a connected valid domain.
For $n \geq 5$, the amplitude $A_k = 1 + \sqrt{2}\cos(\cdot)$ is
negative for some modes at every orientation $\theta$, so the Koide
theorem does not extend directly.

The energy scale of the $n$-th family is given by Faddeev--Niemi scaling:
\begin{equation*}
  M_n \;=\; M_3 \left(\frac{n}{3}\right)^{3/4}\,,
\end{equation*}
where $M_3 = 313.86$~MeV is the trefoil scale from Paper~V.
This gives $M_5 = 460.4$~MeV for the cinquefoil $T(2,5)$,
$M_7 = 592.5$~MeV for the heptafoil $T(2,7)$, and so on.
These are model predictions for the energy scales at which candidate
knotted-vortex particle families would appear if the hierarchy maps to nature.

The mass spectrum of the $T(2,5)$ family requires separate work:
we identify the structural obstruction (no valid $\theta$ domain with all
$A_k \geq 0$ for $n \geq 5$) and state the open problems precisely.

\section{The General $\mathbb{Z}_n$ Brannen Parametrisation}
\label{sec:brannen}

\subsection{Setup}

The $\mathbb{Z}_n$ analogue of the Brannen parametrisation~\cite{Brannen2006}
for the amplitudes of the $n$ modes of a $T(2,n)$ vortex is:
\begin{equation}
  A_k \;=\; 1 + c_n \cos\!\left(\theta + \frac{2\pi k}{n}\right),
  \quad k = 0, 1, \ldots, n-1\,,
  \label{eq:Zn_brannen}
\end{equation}
and the mode masses are $m_k = M A_k^2$.
The coefficient $c_n$ controls the relative amplitude of the $\mathbb{Z}_n$
oscillation.

\subsection{$\mathbb{Z}_n$ Fourier identities}

Two identities hold for the $\mathbb{Z}_n$ phase lattice $\{2\pi k/n\}$:
\begin{align}
  \sum_{k=0}^{n-1} \cos\!\left(\theta + \frac{2\pi k}{n}\right) &= 0
  \label{eq:ident_I} \\
  \sum_{k=0}^{n-1} \cos^2\!\left(\theta + \frac{2\pi k}{n}\right) &= \frac{n}{2}
  \label{eq:ident_II}
\end{align}
for any $\theta$ and any $n \geq 2$.
From~\eqref{eq:ident_I}: $\sum_k A_k = n$ (the sum of amplitudes is the
dimension $n$, independent of $\theta$).
From both identities: $\text{Var}(A_k) = c_n^2/2$.


\section{Theorem 1: The Universal Coefficient $c_n = \sqrt{2}$}
\label{sec:cn}

\begin{theorem}[Universal Brannen coefficient]
  \label{thm:cn}
  The Brannen amplitude coefficient for any $\mathbb{Z}_n$ family is
  $c_n = \sqrt{2}$, uniquely fixed by the maximum-entropy self-consistency
  condition $\mathrm{Var}(A_k) = \overline{A_k}^2$.
\end{theorem}

\begin{proof}
From identities~\eqref{eq:ident_I}--\eqref{eq:ident_II}:
\[
  \overline{A_k} = 1, \qquad \mathrm{Var}(A_k) = c_n^2/2.
\]
The maximum-entropy condition requires $\mathrm{Var}(A_k) = \overline{A_k}^2$:
\[
  c_n^2/2 = 1 \;\implies\; c_n = \sqrt{2}.
\]
Uniqueness: the condition is a quadratic in $c_n$ with unique positive root.
\end{proof}

\paragraph{Physical interpretation.}
The condition $\text{Var}(A_k) = \overline{A_k}^2$ is equivalent to:
the standard deviation of the amplitude distribution equals its mean.
The amplitude can reach zero (nodal line) or $2\overline{A_k} = 2$
(maximum) with equal probability under the $\mathbb{Z}_n$ phase average.
This is the condition of maximal relative uncertainty in the mode
amplitudes --- the vacuum selects the coefficient that neither suppresses
nor privileges any mode over the background.

\paragraph{Numerical verification.}
For any $n \geq 2$ and any test angle $\theta$, the ratio
$\text{Var}(A_k)/\overline{A_k}^2 = 1.000\ldots$ to machine precision
when $c_n = \sqrt{2}$ (Table~1 of the supplementary).


\section{The Generalised Koide Invariant and the Valid Domain}
\label{sec:koide_general}

\subsection{Derivation of $Q_n = 2/n$}

\begin{proposition}[General Koide invariant]
  \label{prop:Qn}
  For $m_k = M A_k^2$ with $A_k$ as in~\eqref{eq:Zn_brannen} and $c_n = \sqrt{2}$,
  the generalised Koide ratio satisfies $Q_n = 2/n$ provided all $A_k \geq 0$.
\end{proposition}

\begin{proof}
Using identity~\eqref{eq:ident_II} with $c_n = \sqrt{2}$:
\[
  \sum_k m_k = M \sum_k A_k^2 = M\!\left(n + c_n^2 \cdot \frac{n}{2}\right) = 2Mn.
\]
When all $A_k \geq 0$: $\sqrt{m_k} = \sqrt{M} A_k$, so
$\sum_k \sqrt{m_k} = \sqrt{M}\sum_k A_k = \sqrt{M}\cdot n$ (by identity~\eqref{eq:ident_I}).
Therefore:
\[
  Q_n \;\equiv\; \frac{\sum_k m_k}{\Bigl(\sum_k \sqrt{m_k}\Bigr)^2}
  \;=\; \frac{2Mn}{M n^2} \;=\; \frac{2}{n}\,.
\]
\end{proof}

\subsection{The valid domain}

The derivation requires all $A_k \geq 0$.
With $c_n = \sqrt{2}$, the amplitude $A_k = 1 + \sqrt{2}\cos(\cdot)$
reaches its minimum of $1 - \sqrt{2} \approx -0.414$ when the cosine
equals $-1$.
For $n = 3$: the minimum over all $k$ reaches $+0.293$ (positive) at
$\theta = 0$, and reaches zero at $\theta = \pm\pi/12 = \pm 15^\circ$.
So $\mathbb{Z}_3$ has a valid domain $\theta \in (-\pi/12, \pi/12)$.
The physical value $\theta = 2/9 \approx 12.73^\circ$ lies inside this domain.

For $n \geq 5$: no connected interval of $\theta$ exists where all $n$
amplitudes are simultaneously positive.

\begin{theorem}[$\mathbb{Z}_3$ uniqueness]
  \label{thm:Z3_unique}
  $\mathbb{Z}_3$ is the unique torus knot family $T(2,n)$ for which:
  \begin{enumerate}[(a)]
    \item $c_n = \sqrt{2}$ (Theorem~\ref{thm:cn}),
    \item $Q_n = 2/n$ is constant on a connected valid domain
      (Proposition~\ref{prop:Qn}),
    \item the self-consistency condition $n\theta_n = Q_n$ fixes $\theta_n$
      within that domain (Paper~V).
  \end{enumerate}
  For $n \geq 5$, property~(b) fails: no connected valid domain exists.
\end{theorem}

\begin{proof}
  Property~(a) holds for all $n$ (Theorem~\ref{thm:cn}).
  Property~(b) requires a connected domain where all $A_k > 0$;
  the table above shows this fails for $n \geq 5$.
  Property~(c) requires (b) as a prerequisite: $n\theta = 2/n$ with
  $\theta \in (-\pi/12, \pi/12)$ gives $\theta_3 = 2/9$, which satisfies
  $\theta_3 < \pi/12$ ($2/9 \approx 0.222 < \pi/12 \approx 0.262$).
  No analogous closure exists for $n \geq 5$.
\end{proof}

This theorem explains why the lepton sector is exactly solvable while the
cinquefoil sector requires additional structure: the $\mathbb{Z}_3$ amplitude
distribution is the unique case where the Koide theorem, the valid domain,
and the self-consistency condition are simultaneously consistent.

separate, so there is no topological barrier to dissolution.
The particle families of CDFT therefore occur only at odd $n$:
$n = 3, 5, 7, 9, \ldots$

\begin{equation}
  M_n \;=\; M_3 \left(\frac{n}{3}\right)^{3/4}\,,

where $M_3 = 313.86$~MeV is the trefoil scale derived in Paper~V

\subsection{Topological selection of the integer skeleton of $\chi^*$}
\label{sec:crt_selection}

The CDFT spectrum contains three structurally distinguished families whose
indices $n = 7, 12, 27$ arise from independent constraints:
\begin{itemize}
  \item $n = 7$: the heptafoil $T(2,7)$, the unique odd-$n$ family whose
    self-consistency attractor at $\theta_7$ produces the smallest
    $|n\theta_n - Q_n|$ residual when $n\theta_n$ is evaluated at
    $\chi^* = 137.035999177$ (Paper~VIII).
  \item $n = 12 = 4 \times 3$: the product of the four spacetime dimensions
    and the three CDFT particle generations.  This is the generation--spacetime
    index, the degree of the lepton--quark multiplet structure.
  \item $n = 27 = 3^3$: the algebraic closure index of the $\mathbb{Z}_3$
    symmetry group (Theorem~\ref{thm:Z3_unique}).  The Koide formula closes
    on a $\mathbb{Z}_3$ lattice of rank $3^3 = 27$.
\end{itemize}

These three integers jointly determine the integer skeleton of $\chi^*$ via
the Chinese Remainder Theorem.  The simultaneous congruence system
\begin{align}
  n &\equiv 4 \pmod{7}   \label{eq:crt7} \\
  n &\equiv 5 \pmod{12}  \label{eq:crt12} \\
  n &\equiv 2 \pmod{27}  \label{eq:crt27}
\end{align}
has the general solution $n = 137 + 1260\,k$ for $k = 0, 1, 2, \ldots$\,.
The first four solutions are $n = 137, 893, 1649, 2405$;
of these, \textbf{137 is the unique prime}: $893 = 19\times 47$, $1649 = 17\times 97$,
$2405 = 5\times 13\times 37$ (the next prime in the class is $n = 3917$).

\begin{proposition}[Integer skeleton of $\chi^*$]
  \label{prop:crt}
  The integer 137 is the unique minimal prime satisfying the three CRT
  congruences~\eqref{eq:crt7}--\eqref{eq:crt27} imposed by the three
  structurally distinguished CDFT knot families $n = 7$, $n = 12$, $n = 27$.
\end{proposition}

The geometric precision of $\chi^*$ is provided by the vortex ring energy
landscape of Paper~I: the energy minimum $\partial E/\partial\chi = 0$ selects
the unique solution $\chi^* = 137.035999177$ to twelve decimal places~\cite{CODATA2022, MujjabiI2026}.
Together these two results give the \textbf{Mujjabi $\chi^*$ Decomposition}:
\begin{equation}
  \chi^* \;=\; \underbrace{137}_{\text{topology (CRT)}}
            \;+\; \underbrace{0.035999177}_{\text{geometry (Paper~I vortex)}}
  \label{eq:chi_decomposition}
\end{equation}
The integer part is topologically protected by the primality constraint of
Proposition~\ref{prop:crt}; the decimal part is the geometric fine-tuning
of the vortex core profile.


\label{sec:cinquefoil}

\subsection{What is established}

\paragraph{Energy scale.}
$M_5 = 460.4$~MeV (Eq.~\eqref{eq:Mn}, zero free parameters).

\paragraph{Topology.}
$T(2,5)$ is a knot, hence topologically stable.
It supports $\mathbb{Z}_5$-symmetric modes (five lobes).

\paragraph{Amplitude coefficient.}
$c_5 = \sqrt{2}$ (Theorem~\ref{thm:cn}).

\paragraph{Hadron scale context.}
$M_5 = 460.4$~MeV lies between the kaon ($\sim 494$~MeV) and the
eta meson ($\sim 548$~MeV).
We do not claim the cinquefoil corresponds to a specific known particle:
the mass spectrum depends on $\theta_5$ (open problem~1 below).

\subsection{The structural obstruction for $n \geq 5$}

With $c_5 = \sqrt{2}$, the five amplitudes
$A_k = 1 + \sqrt{2}\cos(\theta + 2\pi k/5)$ for $k = 0,\ldots,4$
are never all positive simultaneously (see the valid-domain table in
Section~\ref{sec:koide_general}).
The minimum amplitude over all five is always negative,
with maximum value $-0.144$ at $\theta = 0$.

This means:
\begin{enumerate}
  \item $Q_5 = 2/5$ does not hold as a universal constant
    (Proposition~\ref{prop:Qn} fails).
  \item The step $\sum_k \sqrt{m_k} = \sqrt{M}\sum_k A_k$ is invalid
    because some $A_k < 0$ requires $\sqrt{m_k} = \sqrt{M}|A_k|$.
  \item The self-consistency condition $5\theta_5 = Q_5 = 2/5$
    cannot be applied without first establishing the correct $Q_5$.
\end{enumerate}

\subsection{The modified Koide ratio for $n \geq 5$}

When some $A_k < 0$, define $\tilde{Q}_n(\theta)$ using $|A_k|$:
\begin{equation}
  \tilde{Q}_n(\theta) \;\equiv\;
  \frac{\sum_k A_k^2}{\Bigl(\sum_k |A_k|\Bigr)^2}\,.
  \label{eq:Q_tilde}
\end{equation}
For $\mathbb{Z}_3$ in its valid domain: $\tilde{Q}_3 = Q_3 = 2/3$ (constant,
since all $A_k > 0$ there).
For $\mathbb{Z}_5$: $\tilde{Q}_5(\theta)$ varies from $0.294$ to $0.326$,
far from the putative value $2/5 = 0.400$.

The function $\tilde{Q}_5(\theta)$ is not constant.
This means: no analogue of the Koide self-consistency argument from Paper~V

A derivation of $\theta_5$ requires either:
(a) a physical principle beyond the $\mathbb{Z}_n$ algebraic structure alone, or
(b) a modified Brannen parametrisation for the $T(2,5)$ family that avoids

\label{sec:open}

\begin{enumerate}

  (b) a modified amplitude form for $T(2,5)$ that ensures all $A_k \geq 0$.

\item \textbf{The $T(2,5)$ mass spectrum.}
  With $\theta_5$ known, the five mode masses are
  $m_k^{(5)} = M_5 |A_k^{(5)}|^2$.
  Identification of these five modes with known particles
  (at the $\sim 460$~MeV scale: kaons, etas, or new states)
  depends on whether the cinquefoil modes are confined (quark-like)
  or free (lepton-like).

\item \textbf{Why is $\mathbb{Z}_3$ exactly solvable?}
  Theorem~\ref{thm:Z3_unique} identifies the structural reason
  ($n = 3$ uniquely has a valid domain with $c_n = \sqrt{2}$),
  but a deeper question remains: why does the CDFT vacuum select
  a ground state where the first stable knotted excitation ($T(2,3)$)
  is exactly the one where the Koide self-consistency is possible?

\item \textbf{The general $n$-family chirality equilibrium.}
  Paper~III derived $3\theta + \phi_c = \pi$ from the $\mathbb{Z}_3$
  chirality interaction $\lambda_c \cos(3\theta + \phi_c)$.
  The $\mathbb{Z}_5$ analogue would be $5\theta_5 + \phi_{c,5} = \pi$
  (or some other value).
  Whether this generalises directly and what determines $\phi_{c,5}$
  is an open problem.

\end{enumerate}


\section{Conclusion}

This paper extends the CDFT framework beyond the lepton sector by studying
the general $\mathbb{Z}_n$ structure of torus knot families $T(2,n)$.

\textbf{Theorem~1} proves that $c_n = \sqrt{2}$ for all families,
from the maximum-entropy self-consistency condition
$\text{Var}(A_k) = \overline{A_k}^2$.
This is a universal property of the CDFT vacuum that applies across all
particle families.

\textbf{Theorem~2} proves that $\mathbb{Z}_3$ is the unique family where
the Koide invariant $Q_n = 2/n$ is constant on a connected valid domain.
For $n \geq 5$, the amplitude $A_k = 1 + \sqrt{2}\cos(\cdot)$ is
everywhere negative for at least one mode, so the Koide self-consistency
of Paper~V does not extend.
This explains structurally why the lepton sector is exactly solvable and
the cinquefoil sector requires new ingredients.

The Faddeev--Niemi energy hierarchy~\eqref{eq:Mn} gives the first-principles
energy scales of all torus knot families (Table~\ref{tab:hierarchy}),
with $M_5 = 460.4$~MeV the firm prediction for the cinquefoil family.
